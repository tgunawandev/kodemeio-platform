**Goal:** Build `kctl-mm`, a Python/Typer CLI in the `kodemeio-platform` workspace that manages the Mattermost instance at `mm.idtpp.com`, matching full feature parity with `kodemeio-mattermost/scripts/mm-cli.sh`.

**Architecture:** Hybrid transport — `MattermostClient` (subclass of `kctl_lib.api_client.APIClient`) for REST v4 endpoints, and `MMExec` (SSH + `docker compose exec mattermost mmctl --local`) for server-only operations. 24 Typer sub-apps wired through `cli.py` and an `AppContext` from `kctl_lib.callbacks.AppContextBase`.

**Tech Stack:** Python 3.12, Typer, httpx, Pydantic 2, PyYAML, Rich, pytest, pytest-httpx, ruff, mypy strict, hatchling, uv workspace. Depends on `kctl-lib>=0.4.0`.

"""Mattermost REST v4 client."""

from __future__ import annotations

from typing import Any

from kctl_lib.api_client import APIClient
from kctl_lib.exceptions import APIError, AuthenticationError, NotFoundError


class MattermostClient(APIClient):
    """Thin wrapper around Mattermost REST v4."""

    def __init__(self, url: str, token: str, timeout: int = 30) -> None:
        super().__init__(
            base_url=url.rstrip("/") + "/api/v4",
            headers={"Authorization": f"Bearer {token}"},
            timeout=timeout,
        )

    def _raise_for_status(self, status: int, body: dict[str, Any]) -> None:
        msg = body.get("message", "")
        if status == 401:
            raise AuthenticationError(f"Mattermost auth failed: {msg}")
        if status == 404:
            raise NotFoundError(f"Mattermost resource not found: {msg}")
        if status >= 400:
            raise APIError(f"Mattermost API error ({status}): {msg}")

    # users
    def get_me(self) -> dict[str, Any]:
        return self.get("/users/me")

    def get_user_by_username(self, username: str) -> dict[str, Any]:
        return self.get(f"/users/username/{username}")

    def list_users(self, page: int = 0, per_page: int = 60) -> list[dict[str, Any]]:
        return self.get("/users", params={"page": page, "per_page": per_page})

    def create_user(self, payload: dict[str, Any]) -> dict[str, Any]:
        return self.post("/users", json=payload)

    def update_user_active(self, user_id: str, active: bool) -> dict[str, Any]:
        return self.put(f"/users/{user_id}/active", json={"active": active})

    # teams
    def list_teams(self) -> list[dict[str, Any]]:
        return self.get("/teams")

    def get_team_by_name(self, name: str) -> dict[str, Any]:
        return self.get(f"/teams/name/{name}")

```python
"""SSH + docker compose mmctl transport."""

from __future__ import annotations

import json
import shlex
from dataclasses import dataclass
from typing import Any

from kctl_lib.exceptions import CommandError
from kctl_lib.ssh import SSHResult, ssh_run


@dataclass
class MMExec:
    ssh_host: str
    ssh_user: str
    compose_path: str
    compose_service: str = "mattermost"

    def _build(self, args: list[str], json_out: bool) -> str:
        mmctl_args = ["mmctl", "--local"]
        if json_out:
            mmctl_args += ["--format", "json"]
        mmctl_args += args
        inner = " ".join(shlex.quote(a) for a in mmctl_args)
        return (
            f"docker compose -f {shlex.quote(self.compose_path)} "
            f"exec -T {shlex.quote(self.compose_service)} {inner}"
        )

    def mmctl(self, args: list[str]) -> SSHResult:
        cmd = self._build(args, json_out=False)
        result = ssh_run(host=self.ssh_host, user=self.ssh_user, command=cmd)
        if result.returncode != 0:
            raise CommandError(f"mmctl failed (rc={result.returncode}): {result.stderr.strip()}")
        return result

    def mmctl_json(self, args: list[str]) -> Any:
        cmd = self._build(args, json_out=True)
        result = ssh_run(host=self.ssh_host, user=self.ssh_user, command=cmd)
        if result.returncode != 0:
            raise CommandError(f"mmctl failed (rc={result.returncode}): {result.stderr.strip()}")
        return json.loads(result.stdout)

    def docker_compose(self, args: list[str]) -> SSHResult:
        inner = " ".join(shlex.quote(a) for a in args)
        cmd = f"docker compose -f {shlex.quote(self.compose_path)} {inner}"
        result = ssh_run(host=self.ssh_host, user=self.ssh_user, command=cmd)
        if result.returncode != 0:
            raise CommandError(f"compose failed (rc={result.returncode}): {result.stderr.strip()}")
        return result
```

"""AppContext for kctl-mm."""

from __future__ import annotations

from dataclasses import dataclass
from functools import cached_property

from kctl_lib.callbacks import AppContextBase
from kctl_lib.config import Config

from kctl_mm.core.client import MattermostClient
from kctl_mm.core.mm_exec import MMExec

SERVICE_KEY = "mattermost"


@dataclass
class AppContext(AppContextBase):
    service_key: str = SERVICE_KEY

    @cached_property
    def settings(self) -> dict[str, object]:
        cfg = Config.load()
        return cfg.profile(self.service_key, self.profile_name)

    @cached_property
    def client(self) -> MattermostClient:
        s = self.settings
        return MattermostClient(
            url=str(s["url"]),
            token=str(s["token"]),
            timeout=int(s.get("timeout", 30)),
        )

    @cached_property
    def mm_exec(self) -> MMExec:
        s = self.settings
        return MMExec(
            ssh_host=str(s["ssh_host"]),
            ssh_user=str(s.get("ssh_user", "root")),
            compose_path=str(s["compose_path"]),
            compose_service=str(s.get("compose_service", "mattermost")),
        )

```python
"""kctl-mm main CLI entry point."""

from __future__ import annotations

from typing import Annotated

import typer
from kctl_lib import handle_cli_error

from kctl_mm import __version__
from kctl_mm.core.callbacks import AppContext

app = typer.Typer(
    name="kctl-mm",
    help="Kodemeio Mattermost CLI - manage Mattermost Team Edition.",
    no_args_is_help=True,
    rich_markup_mode="rich",
    pretty_exceptions_enable=False,
)


def version_callback(value: bool) -> None:
    if value:
        typer.echo(f"kctl-mm {__version__}")
        raise typer.Exit()


@app.callback()
def main(
    ctx: typer.Context,
    json_output: Annotated[bool, typer.Option("--json")] = False,
    quiet: Annotated[bool, typer.Option("--quiet", "-q")] = False,
    output_format: Annotated[str, typer.Option("--format", "-f")] = "pretty",
    no_header: Annotated[bool, typer.Option("--no-header")] = False,
    profile: Annotated[str | None, typer.Option("--profile", "-p")] = None,
    version: Annotated[
        bool | None, typer.Option("--version", "-V", callback=version_callback, is_eager=True)
    ] = None,
) -> None:
    fmt = "json" if json_output else output_format
    ctx.obj = AppContext(profile_name=profile, format=fmt, quiet=quiet, no_header=no_header)


def _register() -> None:
    from kctl_mm.commands import (
        audit, bots, channels, completions, config_cmd, dashboard, deploy,
        doctor, health, import_export, integrations, jobs, logs, maintenance,
        mm_config, permissions, plugins, posts, self_update, skill_cmd,
        status, teams, users, webhooks,
    )
    mapping = [
        ("config", config_cmd), ("doctor", doctor), ("self-update", self_update),
        ("completions", completions), ("skill", skill_cmd),
        ("status", status), ("logs", logs), ("deploy", deploy),
        ("health", health), ("dashboard", dashboard),
        ("users", users), ("teams", teams), ("channels", channels),
        ("permissions", permissions), ("posts", posts), ("mm-config", mm_config),
        ("maintenance", maintenance), ("webhooks", webhooks), ("bots", bots),
        ("plugins", plugins), ("integrations", integrations), ("jobs", jobs),
        ("audit", audit), ("import-export", import_export),
    ]
    for name, module in mapping:
        app.add_typer(module.app, name=name)


_register()


def _run() -> None:
    try:
        app()
    except Exception as exc:  # noqa: BLE001
        handle_cli_error(exc)


if __name__ == "__main__":
    _run()
```

Note: module is `mm_config` (not `config`) to avoid shadowing Python's `config` name and to make the group name `mm-config` unambiguous.

"""Diagnostic checks."""

from __future__ import annotations

import typer

from kctl_mm.core.callbacks import AppContext

app = typer.Typer(help="Diagnose kctl-mm configuration and connectivity.", invoke_without_command=True)


@app.callback(invoke_without_command=True)
def run(ctx: typer.Context) -> None:
    c: AppContext = ctx.ensure_object(AppContext)
    results: list[tuple[str, bool, str]] = []

    try:
        r = c.client.ping()
        ok = r.get("status") == "OK"
        results.append(("REST ping", ok, f"status={r.get('status')}"))
    except Exception as exc:
        results.append(("REST ping", False, str(exc)))

    try:
        me = c.client.get_me()
        results.append(("REST auth", True, f"authenticated as {me.get('username')}"))
    except Exception as exc:
        results.append(("REST auth", False, str(exc)))

    try:
        r = c.mm_exec.mmctl(["version"])
        results.append(("SSH + mmctl", True, r.stdout.strip().splitlines()[0] if r.stdout else "ok"))
    except Exception as exc:
        results.append(("SSH + mmctl", False, str(exc)))

    for name, ok, detail in results:
        marker = "OK" if ok else "FAIL"
        c.output.raw(f"[{marker}] {name}: {detail}")

    if not all(ok for _, ok, _ in results):
        raise typer.Exit(1)
