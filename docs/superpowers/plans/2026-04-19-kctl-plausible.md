```python
"""Profile management and configuration for kctl-plausible.

Shared config at ~/.config/kodemeio/config.yaml, service-scoped under key
'plausible'. Profile format (4-segment): platform-tenant-stack-app.
"""
from __future__ import annotations

import os
from pathlib import Path
from typing import Any

import yaml
from pydantic import BaseModel

CONFIG_DIR = Path.home() / ".config" / "kodemeio"
CONFIG_FILE = CONFIG_DIR / "config.yaml"

SERVICE_KEY = "plausible"


class ServiceConfig(BaseModel):
    """Service-specific config within a profile."""

    url: str = ""
    api_key: str = ""
    default_site: str = ""


class ConfigFile(BaseModel):
    default_profile: str = "default"
    profiles: dict[str, dict[str, Any]] = {}


def load_raw_config() -> dict:
    if not CONFIG_FILE.exists():
        return {}
    try:
        with open(CONFIG_FILE) as f:
            return yaml.safe_load(f) or {}
    except Exception:
        return {}


def save_raw_config(data: dict) -> None:
    CONFIG_DIR.mkdir(parents=True, exist_ok=True)
    with open(CONFIG_FILE, "w") as f:
        yaml.dump(data, f, default_flow_style=False, sort_keys=False)


def load_config() -> ConfigFile:
    data = load_raw_config()
    return ConfigFile(
        default_profile=data.get("default_profile", "default"),
        profiles=data.get("profiles", {}),
    )


def _is_service_scoped(profile_data: dict) -> bool:
    for _, val in profile_data.items():
        if isinstance(val, dict):
            return True
    return False


def get_service_config(profile_name: str) -> ServiceConfig:
    cfg = load_config()
    profile_data = cfg.profiles.get(profile_name, {})
    if not profile_data:
        return ServiceConfig()
    if _is_service_scoped(profile_data):
        svc_data = profile_data.get(SERVICE_KEY, {})
        if isinstance(svc_data, dict):
            return ServiceConfig(**svc_data)
        return ServiceConfig()
    return ServiceConfig(
        **{k: v for k, v in profile_data.items() if k in ServiceConfig.model_fields}
    )


def set_service_config(profile_name: str, svc_config: ServiceConfig) -> None:
    data = load_raw_config()
    data.setdefault("profiles", {})
    data["profiles"].setdefault(profile_name, {})
    profile = data["profiles"][profile_name]
    if not _is_service_scoped(profile):
        old = dict(profile)
        profile.clear()
        profile[SERVICE_KEY] = old
    svc_data = svc_config.model_dump(exclude_defaults=False)
    for k in list(svc_data.keys()):
        if not svc_data.get(k):
            svc_data.pop(k, None)
    profile[SERVICE_KEY] = svc_data
    save_raw_config(data)


def get_profile_names() -> list[str]:
    return list(load_config().profiles.keys())


def get_default_profile() -> str:
    return load_config().default_profile


def set_default_profile(name: str) -> None:
    data = load_raw_config()
    data["default_profile"] = name
    save_raw_config(data)


def remove_profile(name: str) -> None:
    data = load_raw_config()
    profiles = data.get("profiles", {})
    profiles.pop(name, None)
    if data.get("default_profile") == name:
        data["default_profile"] = next(iter(profiles), "default")
    save_raw_config(data)


def _expand_token(value: str) -> str:
    if value.startswith("${") and value.endswith("}"):
        return os.environ.get(value[2:-1], "")
    return value


def resolve_active_profile_name(profile_name: str | None = None) -> str:
    if profile_name:
        return profile_name
    if env := os.environ.get("KCTL_PLAUSIBLE_PROFILE"):
        return env
    names = get_profile_names()
    if not names:
        raise ValueError(
            "No profile configured. Run: kctl-plausible config init "
            "or pass -p/--profile."
        )
    raise ValueError(
        f"No profile selected. Pass -p/--profile or set KCTL_PLAUSIBLE_PROFILE. "
        f"Available: {', '.join(names)}"
    )


def resolve_connection(
    profile_name: str | None = None,
    url_override: str | None = None,
    api_key_override: str | None = None,
    default_site_override: str | None = None,
) -> tuple[str, str, str]:
    url, api_key, default_site = "", "", ""
    try:
        pname = resolve_active_profile_name(profile_name)
        svc = get_service_config(pname)
        url = svc.url
        api_key = _expand_token(svc.api_key)
        default_site = svc.default_site
    except ValueError:
        pass
    if env_url := os.environ.get("KCTL_PLAUSIBLE_URL"):
        url = env_url
    if env_key := os.environ.get("KCTL_PLAUSIBLE_API_KEY"):
        api_key = env_key
    if env_site := os.environ.get("KCTL_PLAUSIBLE_DEFAULT_SITE"):
        default_site = env_site
    if url_override:
        url = url_override
    if api_key_override:
        api_key = api_key_override
    if default_site_override:
        default_site = default_site_override
    return url, api_key, default_site
```

```python
"""Plausible CE API client."""
from __future__ import annotations

from typing import Any

import httpx
from kctl_lib.api_client import APIClient
from kctl_lib.exceptions import ConfigError


class PlausibleClient(APIClient):
    """Bearer-token client for Plausible CE Stats + Sites Provisioning API."""

    AUTH_HEADER = "Authorization"
    AUTH_PREFIX = "Bearer"
    API_PREFIX = "/api/v1"

    def __init__(
        self,
        base_url: str = "",
        api_key: str = "",
        timeout: float = 30.0,
        max_retries: int = 3,
        **kwargs: Any,
    ):
        if not base_url:
            raise ConfigError("No URL configured. Run: kctl-plausible config init")
        super().__init__(
            base_url=base_url,
            credential=api_key or "unset",
            timeout=timeout,
            retry_enabled=True,
            max_retries=max_retries,
            **kwargs,
        )

    @property
    def root_url(self) -> str:
        return self._base_url.rsplit("/api/v1", 1)[0]

    def check_health(self) -> dict:
        try:
            r = httpx.get(f"{self.root_url}/api/health", timeout=5)
            return r.json() if r.status_code == 200 else {"status": "error"}
        except httpx.HTTPError:
            return {"status": "error", "message": "unreachable"}

    # ---- Sites Provisioning ----
    def list_sites(self) -> list[dict]:
        data = self.get("/sites")
        return data.get("sites", []) if isinstance(data, dict) else data

    def get_site(self, domain: str) -> dict:
        return self.get(f"/sites/{domain}")

    def create_site(self, domain: str, timezone: str = "UTC") -> dict:
        return self.post("/sites", data={"domain": domain, "timezone": timezone})

    def update_site(self, domain: str, **fields: Any) -> dict:
        return self.put(f"/sites/{domain}", data=fields)

    def delete_site(self, domain: str) -> dict:
        return self.delete(f"/sites/{domain}")

    # ---- Goals ----
    def list_goals(self, site_id: str) -> list[dict]:
        data = self.get(f"/sites/goals?site_id={site_id}")
        return data.get("goals", []) if isinstance(data, dict) else data

    def create_goal(self, site_id: str, goal_type: str, event_name: str = "", page_path: str = "") -> dict:
        payload: dict[str, Any] = {"site_id": site_id, "goal_type": goal_type}
        if event_name:
            payload["event_name"] = event_name
        if page_path:
            payload["page_path"] = page_path
        return self.put("/sites/goals", data=payload)

    def delete_goal(self, site_id: str, goal_id: str) -> dict:
        return self.delete(f"/sites/goals/{goal_id}?site_id={site_id}")

    # ---- Funnels ----
    def list_funnels(self, site_id: str) -> list[dict]:
        data = self.get(f"/sites/funnels?site_id={site_id}")
        return data.get("funnels", []) if isinstance(data, dict) else data

    def create_funnel(self, site_id: str, name: str, steps: list[dict]) -> dict:
        return self.put("/sites/funnels", data={"site_id": site_id, "name": name, "steps": steps})

    def delete_funnel(self, site_id: str, funnel_id: str) -> dict:
        return self.delete(f"/sites/funnels/{funnel_id}?site_id={site_id}")

    # ---- Shared Links ----
    def list_shared_links(self, site_id: str) -> list[dict]:
        data = self.get(f"/sites/shared-links?site_id={site_id}")
        return data.get("shared_links", []) if isinstance(data, dict) else data

    def create_shared_link(self, site_id: str, name: str, password: str | None = None) -> dict:
        body: dict[str, Any] = {"site_id": site_id, "name": name}
```

```python
"""AppContext — lazily resolves client + output."""
from __future__ import annotations
from dataclasses import dataclass, field

from kctl_lib.output import Output

from kctl_plausible.core.client import PlausibleClient
from kctl_plausible.core.config import resolve_connection


@dataclass
class AppContext:
    json_mode: bool = False
    quiet: bool = False
    format: str = "pretty"
    no_header: bool = False
    profile: str | None = None
    url_override: str | None = None
    api_key_override: str | None = None
    default_site_override: str | None = None
    _client: PlausibleClient | None = field(default=None, repr=False, init=False)
    _output: Output | None = field(default=None, repr=False, init=False)

    @property
    def output(self) -> Output:
        if self._output is None:
            self._output = Output(
                json_mode=self.json_mode, quiet=self.quiet,
                format=self.format, no_header=self.no_header,
            )
        return self._output

    @property
    def client(self) -> PlausibleClient:
        if self._client is None:
            url, key, site = resolve_connection(
                profile_name=self.profile,
                url_override=self.url_override,
                api_key_override=self.api_key_override,
                default_site_override=self.default_site_override,
            )
            self._client = PlausibleClient(base_url=url, api_key=key)
            self._client.default_site = site  # type: ignore[attr-defined]
        return self._client

    def close(self) -> None:
        if self._client is not None:
            self._client.close()
```

```python
"""Goal / UTM / slug validators (spec §6)."""
from __future__ import annotations
import re

from kctl_lib.exceptions import ValidationError


class GoalNameError(ValidationError):
    """Invalid Plausible goal name."""


class UTMCampaignError(ValidationError):
    """Invalid utm_campaign value."""


GOAL_RE = re.compile(
    r"^(bas|hrm|tpm|tms|agency|careers)\."
    r"(interest|lead|trial|paid|expansion|apply)\."
    r"[a-z][a-z0-9_]*$"
)

UTM_CAMPAIGN_RE = re.compile(
    r"^(?:stg_)?(20\d{2})_q[1-4]_(bas|hrm|tpm|tms|agency|careers|cross)_[a-z0-9_]+$"
)


def validate_goal_name(name: str) -> None:
    if not name or not GOAL_RE.match(name):
        raise GoalNameError(
            f"Invalid goal name {name!r}. Expected "
            "<product>.<stage>.<event>, products: bas|hrm|tpm|tms|agency|careers, "
            "stages: interest|lead|trial|paid|expansion|apply, event: lower snake_case."
        )


def validate_utm_campaign(campaign: str) -> None:
    if not campaign or not UTM_CAMPAIGN_RE.match(campaign):
        raise UTMCampaignError(
            f"Invalid utm_campaign {campaign!r}. Expected "
            "{yyyy}_q{1-4}_{product|cross}_{theme_snake_case}."
        )
```

```python
"""Declarative bootstrap YAML schema for `sites bootstrap`."""
from __future__ import annotations
from pathlib import Path

import yaml
from pydantic import BaseModel, field_validator

from kctl_plausible.core.validators import validate_goal_name


class FunnelStep(BaseModel):
    name: str
    steps: list[str]

    @field_validator("steps")
    @classmethod
    def _check_steps(cls, v: list[str]) -> list[str]:
        for s in v:
            validate_goal_name(s)
        return v


class SharedLinkSpec(BaseModel):
    name: str
    password: str | None = None


class GuestSpec(BaseModel):
    email: str
    role: str = "viewer"


class SiteSpec(BaseModel):
    domain: str
    timezone: str = "UTC"
    public: bool = False
    goals: list[str] = []
    funnels: list[FunnelStep] = []
    shared_links: list[SharedLinkSpec] = []
    guests: list[GuestSpec] = []

    @field_validator("goals")
    @classmethod
    def _check_goals(cls, v: list[str]) -> list[str]:
        for g in v:
            validate_goal_name(g)
        return v


class BootstrapManifest(BaseModel):
    version: int = 1
    sites: list[SiteSpec]


def load_manifest(path: str | Path) -> BootstrapManifest:
    raw = yaml.safe_load(Path(path).read_text())
    return BootstrapManifest(**raw)
```

```python
from pathlib import Path
from kctl_plausible.core.bootstrap_schema import BootstrapManifest, load_manifest


@app.command("bootstrap")
def bootstrap_(
    ctx: typer.Context,
    manifest: Annotated[Path, typer.Argument(help="Bootstrap YAML manifest")],
    dry_run: Annotated[bool, typer.Option("--dry-run", help="Show diff only")] = False,
) -> None:
    """Apply a declarative bootstrap manifest (idempotent)."""
    c: AppContext = ctx.obj
    out = c.output
    m: BootstrapManifest = load_manifest(manifest)
    client = c.client

    existing_sites = {s.get("domain"): s for s in client.list_sites()}
    planned = {"site_create": 0, "goal_create": 0, "funnel_create": 0, "shared_link_create": 0, "guest_invite": 0}

    for site in m.sites:
        if site.domain not in existing_sites:
            planned["site_create"] += 1
            if not dry_run:
                client.create_site(site.domain, timezone=site.timezone)
                out.info(f"  + site {site.domain}")

        existing_goals = {g.get("event_name", g.get("page_path", "")) for g in (client.list_goals(site.domain) if not dry_run or site.domain in existing_sites else [])}
        for goal in site.goals:
            if goal in existing_goals:
                continue
            planned["goal_create"] += 1
            if not dry_run:
                client.create_goal(site.domain, goal_type="event", event_name=goal)
                out.info(f"  + goal {site.domain}/{goal}")

        existing_funnels = {f.get("name") for f in (client.list_funnels(site.domain) if not dry_run or site.domain in existing_sites else [])}
        for f in site.funnels:
            if f.name in existing_funnels:
                continue
            planned["funnel_create"] += 1
            if not dry_run:
                steps_payload = [{"goal": s} for s in f.steps]
                client.create_funnel(site.domain, name=f.name, steps=steps_payload)
                out.info(f"  + funnel {site.domain}/{f.name}")

        existing_links = {l.get("name") for l in (client.list_shared_links(site.domain) if not dry_run or site.domain in existing_sites else [])}
        for link in site.shared_links:
            if link.name in existing_links:
                continue
            planned["shared_link_create"] += 1
            if not dry_run:
                client.create_shared_link(site.domain, name=link.name, password=link.password)
                out.info(f"  + shared-link {site.domain}/{link.name}")

        existing_guests = {g.get("email") for g in (client.list_guests(site.domain) if not dry_run or site.domain in existing_sites else [])}
        for guest in site.guests:
            if guest.email in existing_guests:
                continue
            planned["guest_invite"] += 1
            if not dry_run:
                client.invite_guest(site.domain, email=guest.email, role=guest.role)
                out.info(f"  + guest {site.domain}/{guest.email}")

    label = "Planned" if dry_run else "Applied"
    for k, v in planned.items():
        out.kv(f"{label} {k}", str(v))
```

```python
from __future__ import annotations
from typing import Annotated

import typer

from kctl_plausible.core.callbacks import AppContext

app = typer.Typer(help="Curated reports across products.")

PRODUCT_SITES = {
    "bas": "bas.kodeme.io",
    "hrm": "hrm.kodeme.io",
    "tpm": "provetics.com",
    "tms": "terakidz.com",
}

PRODUCT_FUNNEL_GOALS = {
    p: [f"{p}.interest.pricing_view", f"{p}.lead.demo_request", f"{p}.trial.start", f"{p}.paid.conversion"]
    for p in PRODUCT_SITES
}


@app.command("overview")
def overview_(
    ctx: typer.Context,
    period: Annotated[str, typer.Option("--period")] = "7d",
) -> None:
    """One-row-per-site visitors / pageviews / bounce rate."""
    c: AppContext = ctx.obj
    rows = []
    sites = c.client.list_sites()
    for s in sites:
        d = s.get("domain")
        if not d:
            continue
        agg = c.client.stats_aggregate(d, period=period, metrics="visitors,pageviews,bounce_rate")
        r = agg.get("results", {})
        rows.append([
            d,
            str(r.get("visitors", {}).get("value", 0)),
            str(r.get("pageviews", {}).get("value", 0)),
            str(r.get("bounce_rate", {}).get("value", 0)),
        ])
    c.output.table(
        f"Overview ({period})",
        [("Site", "cyan"), ("Visitors", "green"), ("Pageviews", ""), ("Bounce %", "")],
        rows,
        data_for_json=rows,
    )


@app.command("product")
def product_(
    ctx: typer.Context,
    product: Annotated[str, typer.Argument(help="bas|hrm|tpm|tms")],
    period: Annotated[str, typer.Option("--period")] = "30d",
) -> None:
    """Per-product funnel + UTM source breakdown."""
    c: AppContext = ctx.obj
    if product not in PRODUCT_SITES:
        c.output.error(f"Unknown product {product!r}. Must be one of: {', '.join(PRODUCT_SITES)}")
        raise typer.Exit(1)
    site = PRODUCT_SITES[product]

    # Overview block
    agg = c.client.stats_aggregate(site, period=period, metrics="visitors,pageviews,events")
    c.output.detail(
        f"{product.upper()} ({site}, {period})",
        [(k, str(v.get("value") if isinstance(v, dict) else v)) for k, v in agg.get("results", {}).items()],
        data_for_json=agg,
    )

    # Funnel block
    funnel_rows = []
    for g in PRODUCT_FUNNEL_GOALS[product]:
        r = c.client.stats_aggregate(site, period=period, metrics="events", filters=f"event:goal=={g}")
        value = r.get("results", {}).get("events", {}).get("value", 0)
        funnel_rows.append([g, str(value)])
    c.output.table(f"Funnel ({product})", [("Goal", ""), ("Count", "green")], funnel_rows, data_for_json=funnel_rows)

    # UTM source breakdown
    bd = c.client.stats_breakdown(site, period=period, property_="visit:source", metrics="visitors", limit=10)
    src_rows = [[str(r.get("source", "")), str(r.get("visitors", 0))] for r in bd.get("results", [])]
    c.output.table(
        f"Top sources ({product})",
        [("Source", ""), ("Visitors", "green")],
        src_rows,
        data_for_json=bd,
    )
```

```python
def funnel_(
    ctx: typer.Context,
    site: Annotated[str, typer.Argument()],
```

```python
    ctx: typer.Context,
    site: Annotated[str, typer.Argument()],
    property_: Annotated[str, typer.Option("--property")] = "event:page",
    period: Annotated[str, typer.Option("--period")] = "30d",

        return
    fields = list(results[0].keys())
    with open(output, "w", newline="") as f:
        w = _csv.DictWriter(f, fieldnames=fields)
        w.writeheader()
        for row in results:
            w.writerow(row)
    c.output.success(f"Wrote {len(results)} rows -> {output}")
```

```python
from __future__ import annotations
from dataclasses import dataclass

@dataclass
class ConfigCheck:
    name: str = "Configuration"

    def run(self) -> CheckResult:
        try:
            p = resolve_active_profile_name()
            cfg = get_service_config(p)
            if not cfg.url:
                return CheckResult(self.name, "fail", "No URL", fix_command="kctl-plausible config init")
            return CheckResult(self.name, "ok", f"Profile '{p}' -> {cfg.url}")
        except Exception as e:
            return CheckResult(self.name, "warn", str(e))


@dataclass
class APIReachabilityCheck:
    name: str = "Plausible API"
    client_factory: object = None  # injected via closure

    def run(self) -> CheckResult:
        try:
            client = self.client_factory()  # type: ignore[misc]
            h = client.check_health()
            if h.get("status") == "error":
                return CheckResult(self.name, "fail", h.get("message", "health endpoint failed"))
            return CheckResult(self.name, "ok", f"reachable ({', '.join(f'{k}={v}' for k, v in h.items())})")
        except Exception as e:
            return CheckResult(self.name, "fail", str(e))


@dataclass
class SnippetsCheck:
    """Fetch each tracked site and verify Plausible + Meta Pixel + Google Tag snippets."""

    name: str = "Site snippets"
    client_factory: object = None

    def run(self) -> CheckResult:
        try:
            client = self.client_factory()  # type: ignore[misc]
            sites = client.list_sites()
        except Exception as e:
            return CheckResult(self.name, "warn", f"Could not list sites: {e}")
        if not sites:
            return CheckResult(self.name, "warn", "No sites configured")
        problems: list[str] = []
        for s in sites:
            domain = s.get("domain", "")
            if not domain:
                continue
            url = f"https://{domain}"
            try:
                html = httpx.get(url, timeout=10, follow_redirects=True).text
            except Exception as e:
                problems.append(f"{domain}: fetch failed ({e})")
                continue
            if "plausible.kodeme.io/js/script.js" not in html:
                problems.append(f"{domain}: missing Plausible snippet")
            if "fbq('init'" not in html and "fbq(\"init\"" not in html:
                problems.append(f"{domain}: missing Meta Pixel")
            if "gtag('config'" not in html and "gtag(\"config\"" not in html:
                problems.append(f"{domain}: missing Google Tag")
        if problems:
            return CheckResult(self.name, "warn", "; ".join(problems))
        return CheckResult(self.name, "ok", f"{len(sites)} site(s) have all snippets")


@app.callback(invoke_without_command=True)
def doctor_(ctx: typer.Context) -> None:
    """Run diagnostic checks."""
    c: AppContext = ctx.obj
    checks = [
        PythonCheck(),
        UVCheck(),
        GitCheck(),
        DockerCheck(),
        ConfigCheck(),
        APIReachabilityCheck(client_factory=lambda: c.client),
        SnippetsCheck(client_factory=lambda: c.client),
    ]
    failed = run_doctor(checks, output=c.output)
    if failed:
        raise typer.Exit(1)
```
